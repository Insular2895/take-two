const $ = (selector) => document.querySelector(selector);
const h = (value) => String(value ?? "—").replace(/[&<>'"]/g, (character) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", "'": "&#39;", '"': "&quot;" })[character]);
const money = (value, currency = "EUR") => value === null || value === undefined ? "N/A" : new Intl.NumberFormat("fr-FR", { style: "currency", currency, maximumFractionDigits: 0, signDisplay: "exceptZero" }).format(Number(value));
const amount = (value, currency = "EUR") => value === null || value === undefined ? "N/A" : new Intl.NumberFormat("fr-FR", { style: "currency", currency, maximumFractionDigits: 0 }).format(Number(value));
const percent = (value) => value === null || value === undefined ? "N/A" : new Intl.NumberFormat("fr-FR", { style: "percent", maximumFractionDigits: 1, signDisplay: "exceptZero" }).format(Number(value));
const age = (timestamp) => {
  if (!timestamp) return "N/A";
  const seconds = Math.max(0, Math.round((Date.now() - Date.parse(timestamp)) / 1000));
  if (seconds < 60) return `${seconds} sec ago`;
  if (seconds < 3600) return `${Math.floor(seconds / 60)} min ago`;
  if (seconds < 86400) return `${Math.floor(seconds / 3600)} h ago`;
  return `${Math.floor(seconds / 86400)} d ago`;
};

async function api(path, options = {}) {
  const headers = { "X-Requested-With": "XMLHttpRequest", ...(options.body ? { "Content-Type": "application/json" } : {}), ...(options.headers || {}) };
  if (options.method && options.method !== "GET") headers["X-CSRF-Token"] = state.csrf;
  const response = await fetch(path, { ...options, headers });
  if (response.status === 401) { window.location.assign("/"); throw new Error("ACCESS_SESSION_EXPIRED"); }
  const payload = response.headers.get("Content-Type")?.includes("json") ? await response.json() : null;
  if (!response.ok) { const error = new Error(payload?.error || `HTTP_${response.status}`); error.status = response.status; throw error; }
  return payload;
}

function actionErrorMessage(error) { return actionErrorMessages[error.message] || error.message; }
function reportActionError(error, target = null) {
  if (error.message === "ACTION_CANCELLED") return;
  if (target) target.textContent = actionErrorMessage(error);
  else alert(actionErrorMessage(error));
}

function finishActionPrompt(value, error = null) {
  const pending = state.actionPrompt;
  if (!pending) return;
  state.actionPrompt = null;
  const input = $("#action-password");
  input.value = "";
  $("#action-password-dialog").close();
  if (error) pending.reject(error); else pending.resolve(value);
}

function requestActionPassword(actionLabel, feedback = "") {
  if (state.actionPrompt) return Promise.reject(new Error("ACTION_PASSWORD_PROMPT_ALREADY_OPEN"));
  $("#action-password-title").textContent = actionLabel;
  $("#action-password-error").textContent = feedback;
  $("#action-password").value = "";
  $("#action-password-dialog").showModal();
  $("#action-password").focus();
  return new Promise((resolve, reject) => { state.actionPrompt = { resolve, reject }; });
}

function setEconomicValue(selector, value, format = money) {
  const element = $(selector);
  element.textContent = format(value);
  element.classList.toggle("positive", value !== null && Number(value) > 0);
  element.classList.toggle("negative", value !== null && Number(value) < 0);
}

function dlRows(rows) {
  return rows.map(([term, value, source]) => `<dt>${h(term)}${source ? `<small> · ${h(source)}</small>` : ""}</dt><dd>${h(value)}</dd>`).join("");
}

function renderChart(history) {
  const svg = $("#pnl-chart");
  const cutoff = state.range === "today" ? Date.now() - 86400000 : state.range === "7d" ? Date.now() - 7 * 86400000 : 0;
  const points = history.filter((row) => Date.parse(row.timestamp) >= cutoff).slice().reverse();
  if (!points.length) { svg.innerHTML = `<text x="300" y="110" fill="#9aa8b7" text-anchor="middle">No persisted PnL snapshot</text>`; return; }
  const values = points.flatMap((point) => [Number(point.mtm_pnl), point.liquidation_pnl === null ? Number(point.mtm_pnl) : Number(point.liquidation_pnl)]);
  const minimum = Math.min(...values, 0); const maximum = Math.max(...values, 0); const span = Math.max(maximum - minimum, 1);
  const coordinates = (field) => points.map((point, index) => {
    const value = point[field] === null ? Number(point.mtm_pnl) : Number(point[field]);
    return `${30 + index * 540 / Math.max(points.length - 1, 1)},${195 - (value - minimum) / span * 165}`;
  }).join(" ");
  const zeroY = 195 - (0 - minimum) / span * 165;
  svg.innerHTML = `<line x1="30" y1="${zeroY}" x2="570" y2="${zeroY}" stroke="#293444"/><polyline points="${coordinates("mtm_pnl")}" fill="none" stroke="#71b7ff" stroke-width="3"/><polyline points="${coordinates("liquidation_pnl")}" fill="none" stroke="#41d69a" stroke-width="3"/><text x="35" y="22" fill="#9aa8b7">${h(money(maximum))}</text><text x="35" y="214" fill="#9aa8b7">${h(money(minimum))}</text>`;
}

async function refresh() { try { render(await api("/api/dashboard")); } catch (error) { console.error(error); } }

async function mutation(path, body = {}, headers = {}) { const result = await api(path, { method: "POST", headers, body: JSON.stringify(body) }); await refresh(); return result; }
async function protectedMutation(path, body, actionLabel) {
  let feedback = "";
  while (true) {
    const password = await requestActionPassword(actionLabel, feedback);
    try {
      return await mutation(path, body, { "X-Action-Password": password });
    } catch (error) {
      if (error.message === "ACTION_PASSWORD_INVALID") { feedback = actionErrorMessage(error); continue; }
      throw error;
    }
  }
}

function candidateFilters() {
  const ranges = {};
  document.querySelectorAll("[data-range-field]").forEach((input) => {
    const field = input.dataset.rangeField;
    ranges[field] ||= { min: null, max: null };
    ranges[field][input.dataset.bound] = input.value === "" ? null : Number(input.value);
  });
  return {
    view: state.research.view,
    search: $("#filter-search").value.trim(),
    architectures: [...$("#filter-architecture").selectedOptions].map((option) => option.value),
    budgetStatus: $("#filter-budget").value,
    entryType: $("#filter-entry-type").value,
    freshness: $("#filter-freshness").value,
    expiryType: $("#filter-expiry").value,
    ranges,
  };
}

function renderActiveFilterChips(filters) {
  const chips = [];
  if (filters.search) chips.push(`Search: ${filters.search}`);
  if (filters.architectures.length) chips.push(`Structure: ${filters.architectures.join(", ")}`);
  if (filters.budgetStatus) chips.push(`Budget: ${filters.budgetStatus}`);
  if (filters.entryType) chips.push(`Cash flow: ${filters.entryType}`);
  if (filters.freshness) chips.push("Fresh only");
  if (filters.expiryType) chips.push(`${filters.expiryType} expiry`);
  for (const [metric, bounds] of Object.entries(filters.ranges)) {
    if (bounds.min !== null) chips.push(`${METRIC_METADATA[metric]?.label || metric} ≥ ${bounds.min}`);
    if (bounds.max !== null) chips.push(`${METRIC_METADATA[metric]?.label || metric} ≤ ${bounds.max}`);
  }
  $("#active-filter-chips").innerHTML = chips.map((chip) => `<span>${h(chip)}</span>`).join("");
}

function populateCandidateFilters() {
  const candidates = state.research.candidates;
  const options = (values) => [...new Set(values)].sort().map((value) => `<option value="${h(value)}">${h(value)}</option>`).join("");
  $("#filter-architecture").innerHTML = options(candidates.map((item) => item.architecture));
  $("#filter-budget").innerHTML = `<option value="">Tous</option>${options(candidates.map((item) => item.budget_status))}`;
}

async function loadAnalysis(analysisId) {
  try {
    state.research.selectedAnalysisId = analysisId;
    const payload = await api(`/api/research/analyses/${encodeURIComponent(analysisId)}`);
    state.research.analysis = payload.analysis;
    $("#analysis-status").textContent = researchStatusText(payload.analysis);
    const terminal = ["COMPLETE", "NO_TRADE"].includes(payload.analysis.status);
    $("#candidate-workbench").classList.toggle("hidden", !terminal);
    if (terminal) {
      const candidates = await api(`/api/research/analyses/${encodeURIComponent(analysisId)}/candidates`);
      state.research.candidates = candidates.candidates;
      state.research.compare.clear();
      populateCandidateFilters();
      renderCandidates();
      if (payload.analysis.status === "NO_TRADE") $("#analysis-status").textContent += " · NO_TRADE est le verdict de première classe.";
    }
  } catch (error) { $("#research-error").textContent = error.message; }
}

async function refreshResearch() {
  try {
    const payload = await api("/api/research/analyses");
    state.research.analyses = payload.analyses;
    renderResearchHistory();
    const selected = state.research.selectedAnalysisId || payload.analyses[0]?.analysis_request_id;
    if (selected) await loadAnalysis(selected);
  } catch (error) { $("#research-error").textContent = error.message; }
}

  event.preventDefault();

  const budget = {
    preferred_budget: Number(form.get("preferred_budget")), target_budget: Number(form.get("target_budget")),

  };
  $("#research-error").textContent = "";
  if (![budget.preferred_budget, budget.target_budget, budget.maximum_budget].every(Number.isFinite) || !(budget.preferred_budget <= budget.target_budget && budget.target_budget <= budget.maximum_budget)) {
    $("#research-error").textContent = "Le budget doit respecter préféré ≤ cible ≤ maximum."; return;

    await refreshResearch();

});
